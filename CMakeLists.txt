cmake_minimum_required(VERSION 3.20)
# Set before project() so AppleClang's SDK and availability checks agree.
if((CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
   AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND "$ENV{MACOSX_DEPLOYMENT_TARGET}" STREQUAL "")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
project(HttpLocalServer VERSION 3.5.1 LANGUAGES C CXX)
configure_file(cmake/templates/version.hpp.in generated/version.hpp @ONLY)
# Localized MSVC dependency output can omit generated headers in incremental builds.
set_property(SOURCE src/config.cpp src/config_store.cpp src/http_server.cpp src/main.cpp src/mcp.cpp src/service.cpp src/index.cpp src/mounts.cpp src/collaboration.cpp src/durable.cpp src/storage.cpp
  APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp")

option(HLS_STATIC_RUNTIME "Link compiler runtimes statically" ON)
option(HLS_FULL_STATIC "Build a fully static Linux executable (use a musl toolchain)" OFF)
option(HLS_BUILD_TESTS "Build tests" ON)
option(HLS_INFERENCE_ONLY "Build the portable Magika CPU core and probes only" OFF)
if(HLS_FULL_STATIC AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "HLS_FULL_STATIC requires a Linux toolchain; macOS uses the system runtime")
endif()
set(HLS_BOOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/boost-1.83.0" CACHE PATH "Boost headers")
set(HLS_SQLITE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sqlite-amalgamation-3510200" CACHE PATH "SQLite amalgamation")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
include(cmake/Magika.cmake)
if(HLS_INFERENCE_ONLY)
  return()
endif()
include(cmake/TLS.cmake)
include(cmake/QuickJS.cmake)
include(cmake/WAMR.cmake)

if(NOT EXISTS "${HLS_BOOST_ROOT}/boost/beast.hpp")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost-headers-1.83.0.tar.xz" HLS_BOOST_SHA)
  if(NOT HLS_BOOST_SHA STREQUAL "9e073d635431f223add11bcf52fadf245654c4a8650c3a5e9b74cf941d41db70")
    message(FATAL_ERROR "Boost source archive checksum mismatch")
  endif()
  file(MAKE_DIRECTORY "${HLS_BOOST_ROOT}")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost-headers-1.83.0.tar.xz" DESTINATION "${HLS_BOOST_ROOT}")
endif()
if(NOT EXISTS "${HLS_SQLITE_ROOT}/sqlite3.c")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite-amalgamation-3510200.zip" HLS_SQLITE_SHA)
  if(NOT HLS_SQLITE_SHA STREQUAL "6e2a845a493026bdbad0618b2b5a0cf48584faab47384480ed9f592d912f23ec")
    message(FATAL_ERROR "SQLite source archive checksum mismatch")
  endif()
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite-amalgamation-3510200.zip" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/sqlite-source")
  set(HLS_SQLITE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/sqlite-source/sqlite-amalgamation-3510200" CACHE PATH "SQLite amalgamation" FORCE)
endif()
find_package(Threads REQUIRED)
include(cmake/Assets.cmake)
include(cmake/Previews.cmake)
# Keep generated asset headers explicit: localized compiler dependency output
# must not leave a previously embedded page in incremental builds.
set_property(SOURCE src/web_assets.cpp APPEND PROPERTY OBJECT_DEPENDS
  "${CMAKE_CURRENT_BINARY_DIR}/generated/web_data.hpp"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/icon_data.hpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" HLS_LICENSE_TEXT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" HLS_PROJECT_LICENSE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/THIRD_PARTY_NOTICES.txt" HLS_NOTICES)
string(APPEND HLS_LICENSE_TEXT "\n\n${HLS_PROJECT_LICENSE}\n\n${HLS_NOTICES}")
if(HLS_ENABLE_TLS)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/mbedtls-Apache-2.0.txt" HLS_TLS_LICENSE)
  string(APPEND HLS_LICENSE_TEXT "\n\n===== Mbed TLS 3.6.7 (Apache-2.0 option selected) =====\n${HLS_TLS_LICENSE}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/licenses/mbedtls-Apache-2.0.txt")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/licenses/THIRD_PARTY_NOTICES.txt")
foreach(HLS_NOTICE wamr-LICENSE.txt quickjs-ng-MIT.txt quickjs-ng-components.txt quickjs-ng-Unicode-3.0.txt qrcodegen-MIT.txt fflate-MIT.txt marked-MIT.txt gcc-COPYING3.txt gcc-COPYING.RUNTIME.txt mingw-w64-COPYING.MinGW-w64-runtime.txt.txt winpthreads-COPYING.txt musl-COPYRIGHT.txt)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/${HLS_NOTICE}" HLS_NOTICE_TEXT)
  string(APPEND HLS_LICENSE_TEXT "\n\n===== ${HLS_NOTICE} =====\n${HLS_NOTICE_TEXT}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/licenses/${HLS_NOTICE}")
endforeach()
file(GLOB HLS_PREVIEW_NOTICES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/monaco-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/silurus-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/pdfjs-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/core-js-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/-thednp-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/path2d*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/whatwg-fetch*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/web-streams-polyfill*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/text-encoding-*.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/licenses/babel-preview-MIT.txt")
foreach(HLS_NOTICE_PATH IN LISTS HLS_PREVIEW_NOTICES)
  file(READ "${HLS_NOTICE_PATH}" HLS_NOTICE_TEXT)
  string(APPEND HLS_LICENSE_TEXT "\n\n===== Preview dependency =====\n${HLS_NOTICE_TEXT}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HLS_NOTICE_PATH}")
endforeach()
string(HEX "${HLS_LICENSE_TEXT}" HLS_LICENSE_HEX)
string(REGEX REPLACE ".." "0x\\0," HLS_LICENSE_BYTES "${HLS_LICENSE_HEX}")
configure_file(cmake/templates/licenses.hpp.in generated/licenses.hpp @ONLY)
add_library(hls_sqlite STATIC "${HLS_SQLITE_ROOT}/sqlite3.c")
target_include_directories(hls_sqlite PUBLIC "${HLS_SQLITE_ROOT}")
target_compile_definitions(hls_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_DQS=0 SQLITE_ENABLE_FTS5=1)

file(GLOB HLS_CORE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
list(FILTER HLS_CORE_SOURCES EXCLUDE REGEX "/(main|http_server|magika|cpu_kernels|cpu_dispatch)\\.cpp$")
add_library(hls_core STATIC ${HLS_CORE_SOURCES})
target_include_directories(hls_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(hls_core SYSTEM PUBLIC "${HLS_BOOST_ROOT}")
target_compile_definitions(hls_core PUBLIC BOOST_ALL_NO_LIB BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED BOOST_ASIO_NO_DEPRECATED _FILE_OFFSET_BITS=64)
target_link_libraries(hls_core PUBLIC hls_sqlite hls_inference hls_quickjs hls_wamr Threads::Threads)
if(HLS_ENABLE_TLS)
  target_compile_definitions(hls_core PUBLIC HLS_ENABLE_TLS=1)
  target_link_libraries(hls_core PUBLIC MbedTLS::mbedtls MbedTLS::mbedx509 MbedTLS::mbedcrypto)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(hls_core PUBLIC stdc++fs)
endif()
add_executable(HttpLocalServer src/main.cpp src/http_server.cpp)
target_link_libraries(HttpLocalServer PRIVATE hls_core)

if(WIN32)
  enable_language(RC)
  set(HLS_APP_ICON "${CMAKE_CURRENT_SOURCE_DIR}/resources/branding/favicon.ico")
  configure_file(cmake/templates/app.rc.in generated/app.rc @ONLY)
  target_sources(HttpLocalServer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc" PROPERTY OBJECT_DEPENDS "${HLS_APP_ICON}")
  target_compile_definitions(hls_core PUBLIC _WIN32_WINNT=0x0601 WINVER=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(hls_core PUBLIC ws2_32 mswsock iphlpapi shell32)
else()
  target_link_libraries(hls_sqlite PRIVATE ${CMAKE_DL_LIBS} m)
endif()
if(MSVC)
  target_link_options(HttpLocalServer PRIVATE /SUBSYSTEM:CONSOLE,6.01)
  target_compile_options(hls_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(HttpLocalServer PRIVATE /W4 /utf-8 /permissive-)
  if(HLS_STATIC_RUNTIME)
    set_property(TARGET hls_core hls_sqlite HttpLocalServer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
else()
  target_compile_options(hls_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(HttpLocalServer PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW AND HLS_STATIC_RUNTIME)
    target_link_options(HttpLocalServer PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(HLS_FULL_STATIC)
    target_link_options(HttpLocalServer PRIVATE -static)
  endif()
endif()
if(HLS_BUILD_TESTS)
  enable_testing()
  add_executable(hls_tests tests/core_tests.cpp)
  target_link_libraries(hls_tests PRIVATE hls_core)
  if(MSVC AND HLS_STATIC_RUNTIME)
    target_compile_options(hls_tests PRIVATE /utf-8)
    set_property(TARGET hls_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW AND HLS_STATIC_RUNTIME)
    target_link_options(hls_tests PRIVATE -static)
  elseif(HLS_FULL_STATIC)
    target_link_options(hls_tests PRIVATE -static)
  endif()
  add_test(NAME core COMMAND hls_tests)
  add_executable(hls_write_tests tests/write_tests.cpp)
  # Localized MSVC include output can omit transitive layout dependencies.
  set_property(SOURCE tests/write_tests.cpp APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/service.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/index.hpp")
  target_link_libraries(hls_write_tests PRIVATE hls_core)
  if(MSVC AND HLS_STATIC_RUNTIME)
    target_compile_options(hls_write_tests PRIVATE /utf-8)
    set_property(TARGET hls_write_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW AND HLS_STATIC_RUNTIME OR HLS_FULL_STATIC)
    target_link_options(hls_write_tests PRIVATE -static)
  endif()
  add_test(NAME writes COMMAND hls_write_tests)
endif()
install(TARGETS HttpLocalServer RUNTIME DESTINATION bin)
